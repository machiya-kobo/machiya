# Hister SKIP rules: URLs matching any pattern are never indexed (Go regexp, full URL, one per line,
# no spaces inside a pattern; '#' lines are comments). Loading a list replaces the server's skip list.
# Rules only stop NEW captures: `hister reindex` removes already-indexed pages that now match. Hister
# never indexes its own URLs.

# Tailnet + LAN: internal UIs (password manager, Konbini, Niwa, SearXNG results) never go in from browsing.
# Kura's vault push (Niwa/card URLs) gets in on purpose with metadata.ignore_skip_rules=true per document.
^https?://([^/]*\.)?ts\.net([:/]|$)
^https?://(localhost|127\.|10\.|192\.168\.|172\.(1[6-9]|2[0-9]|3[01])\.|100\.)
^https?://[^/.:]+(:[0-9]+)?(/|$)

^https?://[^/]+/(login|log-in|signin|sign-in|signup|oauth2?|authorize|logout|auth)([/?#]|$)
# Login/account/settings pages at ANY depth, whole path segments only

^https?://[^/]+(/[^?#]*)?/(login|log-in|signin|sign-in|accounts?|settings)([/?#]|$)
